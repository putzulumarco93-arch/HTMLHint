function caricaDocentiDivisi() {

    const ex = document.getElementById("docenteEx");

    storici.innerHTML = "";

    nomi.sort();

    nomi.forEach(nome => {
        // Conta gli anni insegnati
        const anniInsegnati = new Set(dati.filter(d => d[1] === nome).map(d => convertiAnno(d[0])));

        option.textContent = nome;

        if (haClasse2025) {
            attuali.appendChild(option);
        } else if (anniInsegnati.size >= 4) {
            storici.appendChild(option);
        } else {
            ex.appendChild(option);
        }
    });
}

function calcolaOre(materia, annoClasse, sezione){

    const biennio = (annoClasse == 1 || annoClasse == 2);

    const triennio = (annoClasse >= 3);

    const sezioni5oreSeconda = ["A","B","C","D","I","M"];

    let mate = 0;

    let fisica = 0;



    switch(materia){

        case "M":

            if(annoClasse == 1) mate = 5;

            else if(annoClasse == 2) mate = sezioni5oreSeconda.includes(sezione) ? 5 : 4;

            else mate = 4;

        break;

        case "F":

            fisica = biennio ? 2 : 3;

        break;

        case "MF":

            if(annoClasse == 1){ mate=5; fisica=2;}

            else if(annoClasse == 2){ mate=sezioni5oreSeconda.includes(sezione)?5:4; fisica=2;}

            else{ mate=4; fisica=3;}

        break;

        case "MPM": mate = biennio ? 6 : 5; break;

        case "FPF": fisica = biennio ? 3 : 4; break;

        case "MFPMPF": mate=5; fisica=4; break;

        case "MT": mate=7; break;

        case "FT": mate=2; fisica=2; break;

    }



    return {totale: mate+fisica, mate: mate, fisica: fisica};

}

function aggiornaDocentiPiuCondivisiSicuro(docente) {
    const rows = document.querySelectorAll("#tableBody tr");
    const classiIndici = [3,4,5,6,7,8];

    function contaNomiInverse(materieFiltro) {
        let counter = {};
        let ultimoAnnoPerDocente = {};

        rows.forEach(r => {
            const anno = r.cells[0].textContent;

            classiIndici.forEach(i => {
                const cella = r.cells[i];
                if (!cella || !cella.innerHTML.includes("<br>")) return;

                const primaRiga = cella.innerHTML.split("<br>")[0]; // es: "3F=7"
                const annoClasse = parseInt(primaRiga[0]);

                // Determino materia del docente basandomi sul testo colorato della cella
                let materiaDocente;
                const colore = cella.style.color || "";
                if (colore === "rgb(225, 6, 0)" || colore === "#E10600" || colore === "rgb(255, 140, 0)" || colore === "#FF8C00") {
                    materiaDocente = "M";
                } else if (colore === "rgb(0, 168, 255)" || colore === "#00A8FF" || colore === "rgb(0, 51, 204)" || colore === "#0033CC") {
                    materiaDocente = "F";
                } else {
                    materiaDocente = ""; // se non riconosciamo colore ignoriamo
                }

                if (!materieFiltro.includes(materiaDocente)) return;

                // Prendo tutti i nomi sotto la classe condivisa
                const nomi = cella.innerHTML.split("<br>").slice(1).map(n => n.replace(/<[^>]*>/g,"").trim());
                nomi.forEach(nome => {
                    if (!nome) return;
                    counter[nome] = (counter[nome] || 0) + 1;
                    ultimoAnnoPerDocente[nome] = annoClasse;
                });
            });
        });

        let maxVal = 0;
        let maxDocenti = [];
        Object.keys(counter).forEach(n => {
            if (counter[n] > maxVal) {
                maxVal = counter[n];
                maxDocenti = [n];
            } else if (counter[n] === maxVal) {
                maxDocenti.push(n);
            }
        });

        if (maxVal === 0) return "";

        if (maxDocenti.length > 1) {
            maxDocenti.sort((a,b) => ultimoAnnoPerDocente[b] - ultimoAnnoPerDocente[a]);
        }

        return `${maxDocenti[0]} (${maxVal})`;
    }

    // Docente di matematica → classi dove insegna fisica
    const materieFisica = ["F"];
    document.getElementById("docenteMatematica").textContent = contaNomiInverse(materieFisica);

    // Docente di fisica → classi dove insegna matematica
    const materieMatematica = ["M"];
    document.getElementById("docenteFisica").textContent = contaNomiInverse(materieMatematica);
}

function aggiungiDocentiCondivisi(docente) {
    const rows = document.querySelectorAll("#tableBody tr");

    rows.forEach(r => {
        const anno = r.cells[0].textContent;

        for (let i = 3; i <= 8; i++) { // celle classi
            let cella = r.cells[i];
            if (!cella.textContent) continue;

            let match = cella.textContent.match(/^(\d)([A-P])=(\d+)/);
            if (!match) continue;

            let annoClasse = parseInt(match[1]);
            let sezione = match[2];
            let numeroOre = match[3];

            let altriDocenti = dati.filter(d => {
                return d[0] === convertiAnnoCodice(anno) &&
                       d[3] == annoClasse &&
                       d[4] == sezione &&
                       d[1] !== docente;
            }).map(d => d[1]);

            if (altriDocenti.length > 0) {
                let primaRiga = `${annoClasse}${sezione}=${numeroOre}`;

                const coloreClasse = cella.style.color;
                let coloreNome = "black";

                if ((coloreClasse === "#E10600" || coloreClasse === "rgb(225, 6, 0)") && annoClasse >= 3) {
                    coloreNome = "#0033CC";
                } else if ((coloreClasse === "#0033CC" || coloreClasse === "rgb(0, 51, 204)") && annoClasse >= 3) {
                    coloreNome = "#E10600";
                } else if ((coloreClasse === "#00A8FF" || coloreClasse === "rgb(0, 168, 255)") && annoClasse <= 2) {
                    coloreNome = "#FF8C00";
                } else if ((coloreClasse === "#FF8C00" || coloreClasse === "rgb(255, 140, 0)") && annoClasse <= 2) {
                    coloreNome = "#00A8FF";
                }

                // Nomi leggermente più grandi e più marcati
                let nomiHtml = altriDocenti.map(n => `<span style="color:${coloreNome}; font-weight:600; font-size:13px;">${n}</span>`).join("<br>");
                cella.innerHTML = primaRiga + "<br>" + nomiHtml;

                r.style.height = "auto";
                cella.style.lineHeight = "1.2em";
            }
        }
    });
}

function convertiAnnoCodice(annoTesto) {
    // Es: "2015/16" → "1516"
    let parts = annoTesto.split("/");
    return parts[0].slice(-2) + parts[1].slice(-2);
}
function calcolaMassimiPerSezione(){

    const tutteSezioni = ["A","B","C","D","E","F","G","H","I","L","M","N","O","P"];
    let risultato = {};

    tutteSezioni.forEach(sez => {

        let oreDocenti = {};

        dati.forEach(d => {

            const docente = d[1];
            const materia = d[2];
            const annoClasse = parseInt(d[3]);
            const sezione = d[4];

            if(sezione === sez){

                if(!oreDocenti[docente]){
                    oreDocenti[docente] = 0;
                }

                const ore = calcolaOre(materia, annoClasse, sezione);
                oreDocenti[docente] += ore.totale;
            }
        });

        // Trovo il massimo
        let maxOre = 0;
        let docenteMax = null;

        Object.keys(oreDocenti).forEach(doc => {
            if(oreDocenti[doc] > maxOre){
                maxOre = oreDocenti[doc];
                docenteMax = doc;
            }
        });

        risultato[sez] = {
            docente: docenteMax,
            ore: maxOre
        };
    });

    return risultato;
}

function ordinaSezioniPerOre(docente) {

    // tutte le sezioni possibili (come nella prima riga)
    const tutteSezioni = ["A","B","C","D","E","F","G","H","I","L","M","N","O","P"];

    // inizializzo conteggio ore per sezione
    let orePerSezione = {};
    tutteSezioni.forEach(s => orePerSezione[s] = 0);

    // sommo le ore del docente per ogni sezione
    dati.forEach(d => {
        if (d[1] === docente) {
            const sezione = d[4];
            const materia = d[2];
            const annoClasse = parseInt(d[3]);

            const ore = calcolaOre(materia, annoClasse, sezione);
            orePerSezione[sezione] += ore.totale;
        }
    });

    // separo sezioni con ore e senza ore
    let conOre = [];
    let senzaOre = [];

    tutteSezioni.forEach(s => {
        if (orePerSezione[s] > 0) {
            conOre.push(s);
        } else {
            senzaOre.push(s);
        }
    });

    // ordino quelle con ore in ordine decrescente
    conOre.sort((a,b) => orePerSezione[b] - orePerSezione[a]);

    // quelle senza ore restano alfabetiche
    senzaOre.sort();

    return [...conOre, ...senzaOre];
}

            }

    // --- Docente fisica con più classi condivise

docenteFisica.parentNode.replaceChild(newDocenteFisica, docenteFisica);

newDocenteFisica.addEventListener("click", () => {

        if (match) {
            const y = match[1].trim();
            const z = match[2];
            mostraDocentePopup(`Il docente <b>${y}</b> è il docente con cui <b>${docente}</b> ha condiviso più classi di matematica, per un totale di <b>${z}</b> volte`);
        } else {
            mostraDocentePopup(`Il docente <b>${y}</b> è il docente con cui <b>${docente}</b> ha condiviso più classi di fisica`);
        }
    }
});

    // --- RIGA 5: anni, ore, classi ecc.
    const celleRiga5 = ["anniLiceo","numeroOre","oreBiennio","oreTriennio","oreDisp","numeroClassi","numeroClassiQuinte"];
    celleRiga5.forEach(id => {
        const td = document.getElementById(id);

        td.parentNode.replaceChild(newTd, td);

        newTd.addEventListener("click", () => {

            switch(id) {
                case "anniLiceo":
                    mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato per <b>${valore}</b> anni al liceo spano`);
                    break;
                case "numeroOre":
                    mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato per un totale di <b>${valore}</b> ore al liceo spano`);
                    break;
                case "oreBiennio":
                    mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato <b>${valore}</b> ore nel biennio`);
                    break;
                case "oreTriennio":
                    if (valore === 0) mostraDocentePopup(`Il docente <b>${docente}</b> non ha mai insegnato al triennio al liceo spano`);
                    else mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato <b>${valore}</b> ore nel triennio`);
                    break;
                case "oreDisp":
                    mostraDocentePopup(`Il docente <b>${docente}</b> ha integrato le ore di insegnamento con <b>${valore}</b> ore a disposizione/potenziamento`);
                    break;
                case "numeroClassi":
                    mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato in <b>${valore}</b> classi diverse`);
                    break;
                case "numeroClassiQuinte":
                    if (valore === 0) mostraDocentePopup(`Il docente <b>${docente}</b> non ha mai insegnato in una classe 5 del liceo spano`);
                    else mostraDocentePopup(`Il docente <b>${docente}</b> ha insegnato in <b>${valore}</b> classi 5`);
                    break;
            }
        });
    });
}

    rows.forEach(r => {

function calcolaNumeroClassi(docente) {
    const rows = document.querySelectorAll("#tableBody tr");
    const numColonne = 6; // colonne classi 3-8
    let totaleClassi = 0;

    for (let col = 3; col <= 8; col++) {
        let colClasse = [];
        let ultimaClasse = null;
        let ultimoAnnoClasse = 0;

        rows.forEach(r => {
            const cella = r.cells[col];
            if (!cella || !cella.textContent) return;

            let primaRiga = cella.textContent.split("<br>")[0].trim();
            let match = primaRiga.match(/^(\d)([A-Z])/);

            if (!match) return;

            let annoClasse = parseInt(match[1]);
            let sezione = match[2];

            // nuova classe se:
            // - non c'è ancora una classe tracciata
            // - oppure l'anno non è consecutivo della stessa sezione
            if (!ultimaClasse || annoClasse !== ultimoAnnoClasse + 1 || sezione !== ultimaClasse) {
                colClasse.push(sezione + "-" + annoClasse); // identificatore temporaneo
                ultimaClasse = sezione;
            }

            ultimoAnnoClasse = annoClasse;
        });

        totaleClassi += colClasse.length;
    }

    document.getElementById("numeroClassi").textContent = totaleClassi;
}

function caricaDocente(tipo) {
    // --- Determina da quale select prendere il docente
    let select;
    if(tipo === 'attuali') select = document.getElementById("docenteAttuali");
    else if(tipo === 'storici') select = document.getElementById("docenteStorici");
    else if(tipo === 'ex') select = document.getElementById("docenteEx");
    else select = document.getElementById("docenteAttuali"); // default

    const docente = select.value;

    // --- Aggiorna righe sezioni
    aggiornaRigaSezioni(docente);

    // --- Inizializza conteggio programmi per anno
    let programmi = { m: {}, f: {} };
    for (let i = 1; i <= 5; i++) {
        programmi.m[i] = { norm: 0, PM: 0, T: 0 };
        programmi.f[i] = { norm: 0, PF: 0 };
    }

    // --- Costruisci tabella scrollabile
    costruisciTabella();
    impostaAltezzaRighe();
    vaiAdAnnoFinale();

    const rows = document.querySelectorAll("#tableBody tr");

    // --- Variabili per accumuli
    let totaleMate = 0;
    let totaleFisica = 0;
    let totaleBiennio = 0;
    let totaleTriennio = 0;
    let orePerSezione = {};

    let classiContinuita = {};
    let classiUniche = new Set();
    let classiPerAnno = {};

    // --- Ciclo sui dati per il docente selezionato
    dati.sort((a, b) => a[0] - b[0]).forEach(d => {
        if (d[1] === docente) {
            let annoClasse = parseInt(d[3]);
            let sezione = d[4];
            let materia = d[2];

            // Aggiornamento dei programmi per materia/anno
            switch (materia) {
                case "M": programmi.m[annoClasse].norm++; break;
                case "F": programmi.f[annoClasse].norm++; break;
                case "MF": programmi.m[annoClasse].norm++; programmi.f[annoClasse].norm++; break;
                case "MPM": programmi.m[annoClasse].norm++; programmi.m[annoClasse].PM++; break;
                case "FPF": programmi.f[annoClasse].norm++; programmi.f[annoClasse].PF++; break;
                case "MFPMPF": programmi.m[annoClasse].norm++; programmi.m[annoClasse].PM++; programmi.f[annoClasse].norm++; programmi.f[annoClasse].PF++; break;
                case "MT": programmi.m[annoClasse].norm++; programmi.m[annoClasse].T++; break;
                case "FT": programmi.m[annoClasse].norm++; programmi.m[annoClasse].T++; break;
            }

            let anno = convertiAnno(d[0]);
            let ore = calcolaOre(materia, annoClasse, sezione);

            totaleMate += ore.mate;
            totaleFisica += ore.fisica;

            if (annoClasse <= 2) totaleBiennio += ore.totale;
            else totaleTriennio += ore.totale;

            if (!orePerSezione[sezione]) orePerSezione[sezione] = 0;
            orePerSezione[sezione] += ore.totale;

            if (!classiPerAnno[anno]) classiPerAnno[anno] = [];
            classiPerAnno[anno].push({
                annoClasse: annoClasse,
                sezione: sezione,
                materia: materia,
                oreTotali: ore.totale,
                nomeClasse: annoClasse + sezione,
                testo: annoClasse + sezione + "=" + ore.totale
            });

            if (!classiContinuita[sezione]) {
                classiContinuita[sezione] = annoClasse;
                classiUniche.add(sezione + "-" + annoClasse);
            } else {
                let ultimaClasse = classiContinuita[sezione];
                if (annoClasse > ultimaClasse) {
                    classiContinuita[sezione] = annoClasse;
                } else if (annoClasse < ultimaClasse) {
                    classiContinuita[sezione] = annoClasse;
                    classiUniche.add(sezione + "-" + annoClasse);
                }
            }
        }
    });

    // --- Aggiorna anno nascita e pensione
    if (datiAnni[docente]) {
        document.getElementById("annoNascita").textContent = datiAnni[docente].nascita;
        document.getElementById("annoPensione").textContent = datiAnni[docente].pensione || "-";
    } else {
        document.getElementById("annoNascita").textContent = "-";
        document.getElementById("annoPensione").textContent = "-";
    }

    // --- Riempimento colonne della tabella
    let slotPrecedenti = [null, null, null, null, null, null];
    anni.forEach(anno => {
        if (!classiPerAnno[anno]) {
            slotPrecedenti = [null, null, null, null, null, null];
            return;
        }

        let classiAnnoCorrente = classiPerAnno[anno];
        let slotCorrenti = [null, null, null, null, null, null];
        let classiNonPiazzate = [];

        classiAnnoCorrente.forEach(cls => {
            let piazzata = false;
            if (cls.annoClasse > 1) {
                let classePrecedenteDaCercare = (cls.annoClasse - 1) + cls.sezione;
                let indiceTrovato = slotPrecedenti.findIndex(p => p && p.nomeClasse === classePrecedenteDaCercare);
                if (indiceTrovato !== -1 && slotCorrenti[indiceTrovato] === null) {
                    slotCorrenti[indiceTrovato] = cls;
                    piazzata = true;
                }
            }
            if (!piazzata) classiNonPiazzate.push(cls);
        });

        classiNonPiazzate.forEach(cls => {
            let indiceLibero = slotCorrenti.findIndex(s => s === null);
            if (indiceLibero !== -1) slotCorrenti[indiceLibero] = cls;
        });

        slotPrecedenti = [...slotCorrenti];

        rows.forEach(r => {
            if (r.cells[0].textContent === anno) {
                for (let i = 0; i < 6; i++) {
                    let cella = r.cells[i + 3];
                    let classeAssegnata = slotCorrenti[i];
                    if (classeAssegnata) {
                        cella.textContent = classeAssegnata.testo;
                        coloraClasse(cella, classeAssegnata.materia, classeAssegnata.annoClasse);
                    } else {
                        cella.textContent = "";
                        cella.style.fontWeight = "normal";
                        cella.style.fontSize = "12px";
                        cella.style.color = "black";
                    }
                }
            }
        });
    });

    // --- Aggiorna riepilogo ore e classi
    document.getElementById("oreMate").textContent = totaleMate;
    document.getElementById("oreFisica").textContent = totaleFisica;
    document.getElementById("oreBiennio").textContent = totaleBiennio;
    document.getElementById("oreTriennio").textContent = totaleTriennio;

    let anniInsegnati = new Set();
    dati.forEach(d => { if (d[1] === docente) anniInsegnati.add(d[0]); });
    document.getElementById("anniLiceo").textContent = anniInsegnati.size;

    const celleSezioni = document.querySelectorAll(".static-part table tr:nth-child(2) td");
    celleSezioni.forEach((cell, index) => {
        const lettera = document.querySelectorAll(".static-part table tr:nth-child(1) td")[index].textContent;
        cell.textContent = orePerSezione[lettera] || "";
    });

    const righeMini = document.querySelectorAll(".mini-table tr");
    for (let i = 1; i <= 5; i++) {
        let m = programmi.m[i];
let f = programmi.f[i];

let testoM = "";
let testoF = "";

// trasformo T in PM equivalenti
let PMtotali = m.PM + (m.T * 2);

if (m.norm > 0) testoM = m.norm.toString();
if (PMtotali > 0) testoM += "+" + PMtotali + " PM";
        righeMini[0].cells[i - 1].textContent = testoM;
        if (f.norm > 0) testoF = f.norm.toString();
        if (f.PF > 0) testoF += "+" + f.PF + " PF";
        righeMini[1].cells[i - 1].textContent = testoF;
    }

    for (let r = 0; r < righeMini.length; r++) {
        for (let c = 0; c < 5; c++) {
            if (r === 0) righeMini[r].cells[c].style.backgroundColor = (c <= 1 ? "#FFC780" : "#E17070");
            else righeMini[r].cells[c].style.backgroundColor = (c <= 1 ? "#80D4FF" : "#7099E1");
            righeMini[r].cells[c].style.fontWeight = "bold";
            righeMini[r].cells[c].style.color = "black";
        }
    }

    document.getElementById("numeroClassi").textContent = classiUniche.size;
    let numeroClassiQuinte = dati.filter(d => d[1] === docente && parseInt(d[3]) === 5).length;
    document.getElementById("numeroClassiQuinte").textContent = numeroClassiQuinte;

    // --- Calcola ore totali e classi condivise
aggiungiDocentiCondivisi(docente);
aggiungiPopup(select.value);
evidenziaContinuita();
calcolaNumeroClassi(docente);
calcolaOrePerAnno(docente);
aggiornaDocentiPiuCondivisiSicuro(docente);
aggiungiPopupTabella(docente);
aggiungiPopupTabella2(docente);
}

caricaDocentiDivisi();
caricaDocente();
